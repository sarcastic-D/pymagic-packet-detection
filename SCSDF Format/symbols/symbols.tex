{\centering
	\par
	\large \textbf{\underline{LIST OF SYMBOLS}} \par
}\vspace{0.7cm}

\begin{table}[h]
\centering
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{|c|l|}
\hline
\textbf{Symbol} & \textbf{Description} \\
\hline
$H(X)$ & Shannon Entropy of byte sequence $X$ \\
\hline
$p(x_i)$ & Probability of byte value $x_i$ in the payload \\
\hline
$\log_2$ & Logarithm to base 2 (information unit: bits) \\
\hline
$\oplus$ & XOR (Exclusive OR) operation \\
\hline
$\sum$ & Summation operator \\
\hline
$\rightarrow$ & Data flow direction (e.g., Sensor $\rightarrow$ Agent) \\
\hline
$\geq$ & Greater than or equal to (threshold comparison) \\
\hline
$\leq$ & Less than or equal to \\
\hline
$\Delta t$ & Time difference / Latency interval (milliseconds) \\
\hline
$\sigma$ & Standard Deviation of latency measurements \\
\hline
$P_{95}$ & 95th Percentile latency value \\
\hline
$W_f$ & Weight assigned to feature $f$ in risk scoring \\
\hline
$S$ & Cumulative risk score for a packet \\
\hline
$\theta$ & Detection threshold (default: 60) \\
\hline
$|P|$ & Size of payload in bytes \\
\hline
\end{tabular}
\end{table}
